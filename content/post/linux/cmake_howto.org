* CMakeLists.txtを書く

#+begin_src makefile
  # CMakeのバージョンを設定
  cmake_minimum_required(VERSION 3.8)
  # プロジェクト名と使用する言語を設定
  set(APP_NAME hoge)
  project(${APP_NAME} VERSION 0.0 LANGUAGES CXX)

  # C++標準の指定
  set(CMAKE_CXX_STANDARD 17)# 11,14,17(3.8),20(3.12),23(3.20),26(3.25)
  set(CMAKE_CXX_STANDARD_REQUIRED ON)

  add_definitions("-Wall")

  #makeの完全な実行コマンドを表示
  #set(CMAKE_VERBOSE_MAKEFILE 1)

  #srcファイル名を記述。hppは不要
  set(SRC
  main.cpp
  huga.cpp
  gaoo.cpp
  )

  # APP_NAMEという実行ファイルをSRCから作成
  add_executable(${APP_NAME} ${SRC})
#+end_src

* make、実行ファイルの作成、実行

#+begin_src bash
  $ cmake -S . -B build
  $ cmake --build
  $ ./build/hoge
#+end_src

-S : path-to-src
-B : path-to-build
--build <build dir>

buildという名前のdirectoryを作成し、その中で作業を全部してくれる。

<build dir>を作らなくても、このスクリプトで勝手に作成してくれる。\\
makeを叩かなくても、実行ファイルを作成(2行目のcmakeで.oを作成,link)。

src修正時は、buildに入って、makeする(cmakeは不要)

* srcを複数directoryに分割している場合


directoryごとにCMakeLists.txtを用意

#+begin_example
- main.cpp,huga.cpp,gaoo.cpp
  CMakeLists.txt
  subdir
    |- CMakeLists.txt
    |- subf.cpp
    |- subf.hpp
#+end_example


** subdir/CMakeLists.txt

subdirをまとめて1つのLIBにする。

#+begin_src makefile
  cmake_minimum_required(VERSION 3.8)
  set(SRC_SUB subf.cpp)
  set(LIB sub)#subという名前をつける。

  #libraryとして結合するとき
  add_library(${LIB} STATIC ${SRC_SUB})
  #objectとして結合するとき
  add_library(${LIB} OBJECT ${SRC_SUB})
#+end_src

** CMakeLists.txtに以下追記

subdirのLIB(sub)を結合する

*** ライブラリとして結合する場合

#+begin_src makefile
  add_subdirectory(subdir)
  target_link_libraries(${APP_NAME} PRIVATE sub)
#+end_src

*** オブジェクトとして結合する場合

add_executableを修正

#+begin_src makefile
  add_subdirectory(subdir)
  add_executable(test test.cpp $<TARGET_OBJECTS:sub>)
#+end_src
